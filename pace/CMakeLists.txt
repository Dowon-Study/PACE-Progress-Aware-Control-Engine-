cmake_minimum_required(VERSION 3.16)
project(pace C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -O2 -g)

set(MUJOCO_DIR "${CMAKE_SOURCE_DIR}/../third_party/mujoco-3.15.0" CACHE PATH "MuJoCo C 라이브러리 경로")

add_library(pace SHARED src/shm.c src/rt.c src/sim_link.c src/hal_sim_shm.c src/config.c src/refbuf.c src/validator.c src/monitor.c src/refpath.c)

add_library(pace_kin SHARED src/kin.c)
target_include_directories(pace_kin PUBLIC include PRIVATE ${MUJOCO_DIR}/include)
target_link_directories(pace_kin PUBLIC ${MUJOCO_DIR}/lib)
target_link_libraries(pace_kin PUBLIC mujoco m)
set_target_properties(pace_kin PROPERTIES BUILD_RPATH "${MUJOCO_DIR}/lib")
target_include_directories(pace PUBLIC include)
target_link_libraries(pace PUBLIC rt pthread m)

add_executable(pace_control src/pace_control.c)
target_link_libraries(pace_control pace pace_kin)
set_target_properties(pace_control PROPERTIES BUILD_RPATH "${MUJOCO_DIR}/lib")

# ---- 8단계: ros2_control hardware_interface HAL (ROS 2 Jazzy 가 있으면 자동 활성, -DPACE_ROS2=OFF 로 끌 수 있음) ----
option(PACE_ROS2 "ros2_control hardware_interface HAL 백엔드와 MuJoCo 플러그인 빌드" ON)
if(PACE_ROS2)
  list(APPEND CMAKE_PREFIX_PATH "/opt/ros/jazzy")
  find_package(hardware_interface QUIET)
  find_package(pluginlib QUIET)
  find_package(rclcpp QUIET)
  find_package(rclcpp_lifecycle QUIET)
  if(hardware_interface_FOUND AND pluginlib_FOUND AND rclcpp_FOUND AND rclcpp_lifecycle_FOUND)
    message(STATUS "ros2_control HAL: enabled")
    add_library(pace_hal_ros2 SHARED src/hal_ros2.cpp)
    target_link_libraries(pace_hal_ros2 PUBLIC pace hardware_interface::hardware_interface pluginlib::pluginlib rclcpp::rclcpp rclcpp_lifecycle::rclcpp_lifecycle)
    target_compile_definitions(pace_control PRIVATE PACE_HAVE_ROS2)
    target_link_libraries(pace_control pace_hal_ros2)
    set_target_properties(pace_control PROPERTIES LINKER_LANGUAGE CXX BUILD_RPATH "${MUJOCO_DIR}/lib;/opt/ros/jazzy/lib")   # ROS 환경을 source 하지 않아도 실행되도록
    set_target_properties(pace_hal_ros2 PROPERTIES BUILD_RPATH "/opt/ros/jazzy/lib")
    target_link_options(pace_control PRIVATE "-Wl,--disable-new-dtags")   # DT_RPATH: ROS 라이브러리의 간접 의존성까지 /opt/ros/jazzy/lib 에서 찾도록
    target_link_options(pace_hal_ros2 PRIVATE "-Wl,--disable-new-dtags")
    # 플러그인 + ament 인덱스 (pluginlib 가 찾는 자리): build/ros2_prefix → 실행 시 AMENT_PREFIX_PATH 에 추가
    set(ROS2_PREFIX "${CMAKE_BINARY_DIR}/ros2_prefix")
    add_library(pace_mujoco_hardware SHARED ros2/pace_mujoco_hardware.cpp)
    target_link_libraries(pace_mujoco_hardware PUBLIC pace hardware_interface::hardware_interface pluginlib::pluginlib rclcpp_lifecycle::rclcpp_lifecycle)
    set_target_properties(pace_mujoco_hardware PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${ROS2_PREFIX}/lib" BUILD_RPATH "${CMAKE_BINARY_DIR}")
    file(MAKE_DIRECTORY "${ROS2_PREFIX}/share/pace_mujoco_hardware" "${ROS2_PREFIX}/share/ament_index/resource_index/packages" "${ROS2_PREFIX}/share/ament_index/resource_index/hardware_interface__pluginlib__plugin")
    configure_file(ros2/package.xml "${ROS2_PREFIX}/share/pace_mujoco_hardware/package.xml" COPYONLY)
    configure_file(ros2/pace_mujoco_hardware_plugin.xml "${ROS2_PREFIX}/share/pace_mujoco_hardware/pace_mujoco_hardware_plugin.xml" COPYONLY)
    file(WRITE "${ROS2_PREFIX}/share/ament_index/resource_index/packages/pace_mujoco_hardware" "")
    file(WRITE "${ROS2_PREFIX}/share/ament_index/resource_index/hardware_interface__pluginlib__plugin/pace_mujoco_hardware" "share/pace_mujoco_hardware/pace_mujoco_hardware_plugin.xml\n")
  else()
    message(STATUS "ros2_control HAL: ROS 2 Jazzy not found, skipped")
  endif()
endif()

add_executable(pace_sim src/pace_sim.c)

target_link_directories(pace_sim PRIVATE ${MUJOCO_DIR}/lib)
